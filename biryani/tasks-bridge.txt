// Quick capture parsing for the Tasks tab, with no network and no AI:
//  - dates, times, repeats and reminders pulled out of a messy note ("call dad tmrw 5pm", "ps plus ends jan 1 2027")
//  - task vs grocery vs food log, so one capture box handles all three
//  - grocery aisle, store ("soup aldi"), and duplicate detection ("pickles" = "dill pickle spears")
// Pure functions: tools/test_taskparse.js runs them in node.

const pad = n => String(n).padStart(2, '0');
const ymd = d => `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`;
const at = (d, k) => { const x = new Date(d.getFullYear(), d.getMonth(), d.getDate() + k, 12); return x; };
const MONTHS = ['jan', 'feb', 'mar', 'apr', 'may', 'jun', 'jul', 'aug', 'sep', 'oct', 'nov', 'dec'];
const MON_RE = '(jan(?:uary)?|feb(?:ruary)?|mar(?:ch)?|apr(?:il)?|may|june?|july?|aug(?:ust)?|sept?(?:ember)?|oct(?:ober)?|nov(?:ember)?|dec(?:ember)?)';
const DAYS = ['sun', 'mon', 'tue', 'wed', 'thu', 'fri', 'sat'];
// Sun / Sat / Wed alone are ordinary words ("sun glasses", "sat down"), so those need the full name.
const DAY_RE = '(sunday|mon(?:day)?|tue(?:s|sday)?|weds|wednesday|thu(?:r|rs|rsday)?|fri(?:day)?|saturday)';
const dayIdx = w => DAYS.indexOf(w.slice(0, 3).toLowerCase());
const monIdx = w => MONTHS.indexOf(w.slice(0, 3).toLowerCase());
const NUMW = { one: 1, two: 2, three: 3, four: 4, five: 5, six: 6, seven: 7, eight: 8, nine: 9, ten: 10, a: 1, an: 1, couple: 2, few: 3 };
const num = w => (/^\d+$/.test(w) ? +w : NUMW[w]);

// Next date with weekday `wd`, today included unless `skipToday`.
function nextDow(now, wd, skipToday = false) {
  let k = (wd - now.getDay() + 7) % 7;
  if (k === 0 && skipToday) k = 7;
  return at(now, k);
}
// "next friday" = the Friday of next week (weeks start Monday).
function nextWeekDow(now, wd) {
  const toMon = (8 - now.getDay()) % 7 || 7; // days until next Monday
  return at(now, toMon + ((wd + 6) % 7));
}
function monthDay(now, m, d, y) {
  if (y != null) return new Date(y < 100 ? 2000 + y : y, m, d, 12);
  let x = new Date(now.getFullYear(), m, d, 12);
  if (ymd(x) < ymd(now)) x = new Date(now.getFullYear() + 1, m, d, 12);
  return x;
}
const validDay = (m, d) => d >= 1 && d <= [31, 29, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31][m];

const PART = { morning: '09:00', afternoon: '14:00', evening: '18:00', night: '21:00' };
const COMPILED = RULES.map(([src, fn]) => [new RegExp(B + src + E, 'i'), fn]);

// "Call dad tomorrow at 5pm" -> { title: 'Call dad', due: '2026-11-01', time: '17:00', rep: null, spans: [...] }
// `spans` are the [start, end) slices of the original text that were read as a date (for highlighting).
function parseWhen(text, now = new Date()) {
  const r = {}, spans = [];
  let work = text;
  for (const [re, fn] of COMPILED) {
    const m = re.exec(work);
    if (!m) continue;
    if (fn(m, r, now) === false) continue;
    spans.push([m.index, m.index + m[0].length]);
    // Blank the match out (same length) so later rules can't reuse it and spans stay aligned.
    work = work.slice(0, m.index) + ' '.repeat(m[0].length) + work.slice(m.index + m[0].length);
  }
  if (r.rep && !r.date) r.date = r.rep.unit === 'day' ? at(now, 0) : r.rep.unit === 'week' ? at(now, 0) : at(now, 0);
  if (r.time && !r.date) {
    // A bare time is today, or tomorrow once it has passed.
    const [h, mi] = r.time.split(':').map(Number);
    r.date = h * 60 + mi > now.getHours() * 60 + now.getMinutes() ? at(now, 0) : at(now, 1);
  }
  // A dash left hanging where a date was cut out: "haywoods - oct 7 ask wife" -> "haywoods ask wife"
  let title = work.replace(/\s*[-–]\s{2,}|\s{2,}[-–]\s*/g, '  ').replace(/\s+/g, ' ').replace(/\s+([,;?!])/g, '$1')
    .replace(/\s+(?:on|at|by|due|this|next|in|for|the|-|–|,)\s*$/i, '')
    .replace(/^\s*(?:on|at|by|due|-|–|,)\s+/i, '')
    .replace(/^(?:remind\s+me\s+(?:to\s+)?|reminder\s*:?\s*(?:to\s+)?|don'?t\s+forget\s+(?:to\s+)?|todo\s*:?\s*)/i, '')
    .replace(/\s*[-–,:;]\s*$/, '').replace(/^\s*[-–,:;]\s*/, '').trim();
  if (!title) title = text.trim();
  return { title, due: r.date ? ymd(r.date) : null, time: r.time || null, rep: r.rep || null, spans };
}

// Moves a repeating task to its next date after `from` (YYYY-MM-DD).
function nextDue(due, rep, from = due) {
  let d = new Date(due + 'T12:00');
  const step = () => {
    if (rep.unit === 'day') d = at(d, rep.n);
    else if (rep.unit === 'week') d = at(d, 7 * rep.n);
    else if (rep.unit === 'month') d = new Date(d.getFullYear(), d.getMonth() + rep.n, d.getDate(), 12);
    else d = new Date(d.getFullYear() + rep.n, d.getMonth(), d.getDate(), 12);
  };
  step();
  while (ymd(d) <= from) step();
  return ymd(d);
}
const repText = rep => !rep ? '' : rep.n === 1 ? { day: 'Daily', week: 'Weekly', month: 'Monthly', year: 'Yearly' }[rep.unit] : `Every ${rep.n} ${rep.unit}s`;

const STOP = new Set('a an the some more of for and or to get buy pick up grab need needs new another extra few couple pack packs bag bags box boxes bottle bottles jar jars can cans lb lbs oz pound pounds g gram grams kg gallon gallons dozen of from at x'.split(' '));
function singular(w) {
  if (w.length <= 3 || /(ss|us|is|ous)$/.test(w)) return w;
  if (/ies$/.test(w)) return w.slice(0, -3) + 'y';
  if (/(ches|shes|xes|oes|sses)$/.test(w)) return w.slice(0, -2);
  if (/s$/.test(w)) return w.slice(0, -1);
  return w;
}
const words = s => s.toLowerCase().replace(/['’]/g, '').replace(/[^a-z0-9\- ]+/g, ' ').split(/\s+/).filter(Boolean);
// Core words for matching: no amounts, filler or plurals. "2 bags of Dill Pickle Spears" -> ['dill','pickle','spear']
const core = s => words(s).filter(w => !STOP.has(w) && !/^\d/.test(w) && !/^\d*x\d*$/.test(w)).map(singular);

function aisleOf(name, learned = {}) {
  const k = core(name).join(' ');
  if (learned[k]) return learned[k];
  const low = name.toLowerCase();
  for (const [re, a] of PHRASES) if (re.test(low)) return a;
  const ws = core(name);
  // The last food word usually names the thing ("greek yogurt", "dill pickle spears" -> pickle).
  for (let i = ws.length - 1; i >= 0; i--) { const a = aisleWord(ws[i]); if (a) return a; }
  return 'other';
}

// Pulls a quantity and store out of a grocery line. "2x enoki mushrooms (~$1.99/ea) aldi" ->
// { name: 'Enoki mushrooms', qty: '2', store: 'aldi' }
function groceryBits(line) {
  let s = line.replace(/\(.*?\)/g, ' ').replace(/~?\$\d+(\.\d+)?(\/\w+)?/g, ' ');
  let store = null;
  for (const [k, , re] of STORES) if (re.test(s.toLowerCase())) { store = k; s = s.replace(new RegExp(`(?:\\s+(?:at|from|@)\\s+)?${re.source}`, 'i'), ' '); break; }
  let qty = null;
  const q = /^\s*(\d+(?:\.\d+)?)\s*(?:x\b|×)?\s*/i.exec(s) || /\s+x?\s*(\d+)\s*$/i.exec(s);
  if (q && !/^\s*\d+(?:\.\d+)?\s*(?:%|(?:lb|lbs|oz|g|kg|ml|l|gal|pct|percent)\b)/i.test(s.slice(q.index).trimStart() ? s.slice(q.index) : '')) { qty = q[1]; s = s.slice(0, q.index) + ' ' + s.slice(q.index + q[0].length); }
  s = s.replace(/^\s*(?:buy|get|grab|pick\s+up|need|more)\s+/i, '').replace(/\s+/g, ' ').replace(/^[\s,.\-:]+|[\s,.\-:]+$/g, '');
  return { name: s ? s[0].toUpperCase() + s.slice(1) : '', qty, store };
}

// Same item? "pickles" vs "dill pickle spears", "ff cottage cheese" vs "cottage cheese".
// One name's core words all appear in the other's, and they share a food word (not just "cheese" vs "cheese").
function sameItem(a, b) {
  const A = core(a), Bw = core(b);
  if (!A.length || !Bw.length) return false;
  if (A.join(' ') === Bw.join(' ')) return true;
  const [small, big] = A.length <= Bw.length ? [A, Bw] : [Bw, A];
  const bigSet = new Set(big);
  if (!small.every(w => bigSet.has(w))) return false;
  // A lone generic word only matches when it's the head noun ("cheese" != "feta cheese" but "pickle" ~ "dill pickle spear")
  if (small.length === 1 && ['cheese', 'sauce', 'milk', 'bread', 'oil', 'juice', 'water', 'powder', 'butter', 'cream'].includes(small[0])) return big.length === 1;
  return true;
}

const TASK_VERB = /^(?:call|text|email|e-mail|reply|respond|message|msg|dm|ask|tell|remind|pay|cancel|book|schedule|reschedule|apply|submit|file|renew|return|exchange|send|mail|ship|drop\s+off|pick\s+up\s+(?:the\s+)?(?:kids|package|order|prescription|dry)|fix|clean|wash|vacuum|mop|laundry|do|finish|start|check|review|read|watch|update|sign|print|scan|post|sell|list|meet|visit|go|take|bring|move|change|set\s+up|setup|install|research|look\s+into|find|figure|plan|prep|make|write|study|practice|register|confirm|follow\s+up|contact|request|order|transfer|deposit|withdraw|backup|back\s+up|clear|organize|charge|replace|repair|install|rsvp|zelle|venmo)\b/i;
const NOT_GROCERY = /\b(appt|appointment|meeting|interview|call|dentist|doctor|bill|bills|payment|lease|rent|insurance|tax|taxes|passport|visa|license|registration|flight|hotel|ticket|resume|linkedin|job|work|gift|birthday|bday|party|wedding|event|events|claude|reddit|email|movie|show|anime|episode|season|book|course|class|order|return|pickup|refund|credit|account|subscription|plan|annual|renewal|renewalis|ends|expire|expires|deadline|due)\b/i;
// Things around the house that share words with food ("water dispenser", "ice maker").
const NOT_GROCERY_THING = /\b(?:dispenser|icemaker|icemake|maker|machine|fridge|freezer|oven|stove|car|phone|laptop|tv|filter change|repair|broken|leak)\b/i;
const FOOD_LOG = /^(?:ate|had|log|logged|eating)\b|\b\d+(?:\.\d+)?\s*(?:g|grams?|oz|ounces?|ml)\b/i;
// Weights jotted down to log later: "204 pasta 70 craft 80 cottage cheese" (two or more 20+ numbers before words).
const WEIGHTS = /(?:\b(?:[2-9]\d|\d{3,4})\s+[a-z][^\d,]*[\s,]*){2,}/i;
const isFoodLog = s => (FOOD_LOG.test(s) || WEIGHTS.test(s)) && !TASK_VERB.test(s) && !/\$|\bmiles?\b|\bk\b/i.test(s);
const WATCH = /\b(?:watch|movie|film|show|series|anime|season|episode|ghibli|netflix|hulu|crunchyroll|read|book|novel|podcast|listen)\b/i;

// One line of a capture -> 'food' | 'grocery' | 'task'. `known` = Set of your own food names (lowercased core words).
function kindOf(line, { known = new Set(), learned = {} } = {}) {
  const low = line.toLowerCase().trim();
  if (!low) return 'task';
  const k = core(line).join(' ');
  if (learned[k]) return learned[k] === 'task' ? 'task' : 'grocery';
  if (/^(?:buy|get|grab|need|more|out of)\b/.test(low) && !NOT_GROCERY.test(low)) return 'grocery';
  if (isFoodLog(low)) return 'food';
  if (TASK_VERB.test(low) || NOT_GROCERY.test(low)) return 'task';
  const g = groceryBits(line);
  const ws = core(g.name || line);
  if (!ws.length) return 'task';
  if (known.has(ws.join(' '))) return 'grocery';
  const hits = ws.filter(aisleWord).length;
  for (const [re] of PHRASES) if (re.test(low)) return 'grocery';
  // Short lines that are mostly food words: "dill pickle spears", "tums", "deli tuna", "soup aldi"
  if (hits && hits / ws.length >= 0.5 && ws.length <= 5) return 'grocery';
  // Brand + product: "hikari miso paste", "beyond meat nuggets"
  if (hits && ws.length <= 3 && !NOT_GROCERY_THING.test(low)) return 'grocery';
  if (g.store && hits) return 'grocery';
  return 'task';
}

// ---------- splitting a brain dump ----------
// One capture can hold several things: "call dentist, milk, eggs, idea: grokbot, watch dune, pay rent friday".
// Each piece is sorted on its own (task / grocery / food, then list and category in addItem).
// Splits on new lines, ";", commas, sentence ends (". " "? " "! "), and spoken joins ("also", "and then").
// A header carries over to every piece: "Claude code - use cases, gmail export" -> "Claude code: use cases", ...
// "groceries: …", "ideas: …", "tomorrow: …" and a store ("aldi: …") are understood; a header alone on a line
// ("Claude code:") applies to the lines under it. Pieces that only continue the one before stay attached
// ("…, but only if it's on sale", "…, by friday", "…, 2026").
const ABBR = /(?:\b(?:dr|mr|mrs|ms|st|vs|jr|sr|mt|ft|etc|approx|no|inc|co|e\.g|i\.e)|(?:^|\s)[a-z])\.$/i;
const CONT = /^(?:but|because|cause|cuz|so|which|that|who|whose|where|while|if|since|though|although|or|nor|he|she|it|its|it's|they|them|their|him|her|his|with|without|for|from|of|than|as|like|etc|aka|including|especially|not|to(?!\s+do\b))\b/i;
const JOIN = /\s+(?:and\s+also|also|and\s+then|oh\s+and|oh\s+also|another\s+thing(?:\s+is)?)\s+(?=\S)/gi;
// Spoken dumps have no commas: "pay rent friday call mom tomorrow buy milk". A new one starts at one of these verbs.
const STRONG = /\s(?:and\s+|then\s+)?(?=(?:call|text|email|pay|buy|cancel|book|schedule|return|renew|submit|order|pick\s+up|drop\s+off|remind\s+me|send)\s)/gi;
const LINKER = /\b(?:to|or|the|a|an|will|should|need|must|gotta|have|not|dont|don't|please|me|i|can|could|would|and|then|also)$/i;
const wc = s => s.trim().split(/\s+/).filter(Boolean).length;
const hasWords = s => /[a-z]{2,}/i.test(s);
// Only a date or time ("friday", "tomorrow 5pm", "2026")?
const dateOnly = s => { if (!hasWords(s)) return true; const w = parseWhen(s); if (!w.spans.length) return false; let t = s; for (const [a, b] of w.spans) t = t.slice(0, a) + ' '.repeat(b - a) + t.slice(b); return !hasWords(t.replace(/\b(?:on|at|by|due|this|next|the|in)\b/gi, '')); };

function sentences(line) {
  const out = [];
  for (const s of line.split(/(?<=[^\s.?!][.?!]+)\s+(?=\S)/)) {
    if (out.length && ABBR.test(out[out.length - 1])) out[out.length - 1] += ' ' + s; else out.push(s);
  }
  return out.map(s => s.replace(/\.+$/, '').trim()).filter(Boolean);
}
// Splits `s` at every match of `re` where both sides look like their own item.
function splitWhere(s, re, ok) {
  const out = []; let from = 0;
  for (const m of s.matchAll(re)) {
    const left = s.slice(from, m.index).trim(), right = s.slice(m.index + m[0].length).trim();
    if (ok(left, right)) { out.push(left); from = m.index + m[0].length; }
  }
  out.push(s.slice(from).trim());
  return out.filter(Boolean);
}
// A line -> its pieces (strings), before deciding whether to keep them apart.
function piecesOf(body, ctx) {
  const out = [];
  for (const sen of sentences(body)) {
    let parts = sen.split(/\s*(?:;|,(?!\d)|\s+&\s+)\s*/).map(p => p.trim()).filter(Boolean);
    parts = parts.flatMap(p => splitWhere(p, JOIN, (l, r) => wc(l) >= 2 && wc(r) >= 2 && (TASK_VERB.test(r) || /^(?:buy|get|grab)\b/i.test(r))));
    if (parts.length === 1 && wc(sen) >= 6) parts = splitWhere(sen, STRONG, (l, r) => wc(l) >= 2 && wc(r) >= 2 && !LINKER.test(l));
    parts.forEach((p, i) => {
      p = p.replace(/^(?:and|then|plus)\s+/i, '').replace(/\s+(?:and|then)$/i, '').trim();
      if (!p) return;
      // "…, by friday" / "…, but only on sale" / "…, 2026" / ". She's sick." belong to the piece before.
      if (out.length && (CONT.test(p) || dateOnly(p))) out[out.length - 1] += (i === 0 ? '. ' : dateOnly(p) ? ' ' : ', ') + p;
      else out.push(p);
    });
  }
  return out;
}

const HEAD = /^([^,;:]{2,40}?)(?<!\d)\s*:\s+(\S.*)$|^([^,;]{2,40}?)\s+[-–—]+\s+(\S.*)$/;
const HEAD_ALONE = /^([^,;:]{2,40}?)\s*[:\-–—]\s*$/;
function headType(h) {
  const low = h.toLowerCase().trim();
  if (/^(?:groceries|grocery(?:\s+(?:list|run))?)$/.test(low)) return 'groc';
  if (/^(?:shopping(?:\s+list)?|to\s+buy|buy|get|need|pick\s+up|grab)$/.test(low)) return 'buy';
  if (/^(?:ideas?|thoughts?|maybe|brainstorm)$/.test(low)) return 'idea';
  if (/^(?:to\s*-?\s*dos?|tasks?|reminders?)$/.test(low)) return 'plain';
  if (STORES.some(([, , re]) => re.test(low) && !hasWords(low.replace(re, '').replace(/\b(?:at|from|run|trip)\b/g, '')))) return 'store';
  if (dateOnly(h)) return 'date';
  return 'generic';
}
// One piece under a header -> { text, kind }.
function underHead(p, h, type, ctx) {
  const k = kindOf(p, ctx);
  switch (type) {
    case 'groc': return { text: p, kind: TASK_VERB.test(p) ? 'task' : 'grocery' };
    case 'buy': return k === 'grocery' || TASK_VERB.test(p) || /^buy\b/i.test(p) ? { text: p, kind: k } : { text: 'Buy ' + p, kind: 'task' };
    case 'idea': return { text: 'idea: ' + p, kind: 'task' };
    case 'store': return k === 'grocery' ? { text: `${p} ${h}`, kind: k } : { text: `${cap1(h)}: ${p}`, kind: k };
    case 'date': return k === 'grocery' ? { text: p, kind: k } : { text: `${p} ${h}`, kind: k };
    case 'generic': return k === 'task' ? { text: `${cap1(h)}: ${p}`, kind: k } : { text: p, kind: k };
    default: return { text: p, kind: k };
  }
}
const cap1 = s => s.trim()[0].toUpperCase() + s.trim().slice(1);

// Without a header, should these pieces be separate items?
function keepApart(ps, kinds) {
  if (ps.length < 2) return false;
  if (ps.length >= 3) return ps.filter(p => wc(p) <= 12).length >= Math.ceil(ps.length * 2 / 3);
  const [a, b] = ps;
  return kinds[0] !== kinds[1] || listFor(a) !== listFor(b) || (TASK_VERB.test(a) && TASK_VERB.test(b));
}

// Splits a capture into items: [{ text, kind, line }] (`line` = which line of the capture it came from).
// ctx.noSplit keeps each line whole ("Keep as one" in the preview).
function splitCapture(text, ctx = {}) {
  const out = [];
  const lines = text.split(/\n+/).map(l => l.trim().replace(/^\s*(?:[-*•]|\d+[.)])\s+/, '')).filter(Boolean);
  let carry = null; // a header alone on a line applies to the lines under it
  lines.forEach((raw, li) => {
    const push = (t, kind) => { t = t.trim().replace(/[\s,;]+$/, ''); if (t) out.push({ text: t, kind, line: li }); };
    if (ctx.noSplit) return push(raw, kindOf(raw, ctx));
    const alone = HEAD_ALONE.exec(raw);
    if (alone && li < lines.length - 1 && wc(alone[1]) <= 5) { carry = alone[1].trim(); return; }
    if (REF_PREFIX.test(raw)) return push(raw, 'task'); // notes stay whole
    let head = carry, body = raw;
    const hm = HEAD.exec(raw);
    if (hm) {
      const h = (hm[1] || hm[3]).trim(), b = (hm[2] || hm[4]).trim();
      const type = headType(h);
      if (wc(h) <= 5 && (type !== 'generic' || piecesOf(b, ctx).length > 1) && !(type === 'date' && !piecesOf(b, ctx).length)) { head = h; body = b; }
    }
    const ps = piecesOf(body, ctx);
    if (head) {
      const type = headType(head);
      // An idea or note header keeps a one-piece line as is; everything else applies to each piece.
      ps.forEach(p => { const it = underHead(p, head, type, ctx); push(it.text, it.kind); });
      return;
    }
    const kinds = ps.map(p => kindOf(p, ctx));
    // Food log ("chicken 8.5 oz, rice 150 g") stays one line for the food logger, unless a task is mixed in.
    if (isFoodLog(body) && kindOf(body, ctx) === 'food' && !kinds.includes('task')) return push(body, 'food');
    const g = kinds.filter(k => k === 'grocery').length;
    if (ps.length > 1 && g >= Math.ceil(ps.length / 2)) {
      // "and" inside a grocery part: "eggs and milk"
      ps.forEach((p, i) => (kinds[i] === 'grocery' ? p.split(/\s+and\s+/).filter(Boolean) : [p]).forEach(s => push(s, kinds[i] === 'grocery' ? 'grocery' : kindOf(s, ctx))));
      return;
    }
    if (keepApart(ps, kinds)) {
      const food = isFoodLog(body);
      // Mostly ideas ("idea section, connect to wii, grokbot"): a plain note in the same breath is an idea too.
      const ideas = ps.filter(p => listFor(p) === 'ideas').length >= ps.length / 2;
      ps.forEach((p, i) => {
        const plain = ideas && kinds[i] === 'task' && listFor(p) === 'todo' && !TASK_VERB.test(p) && !parseWhen(p).due;
        push(plain ? 'idea: ' + p : p, food && kinds[i] !== 'task' ? 'food' : kinds[i]);
      });
      return;
    }
    const kind = kindOf(body, ctx);
    // "milk and eggs" -> two groceries, when both halves are groceries on their own.
    const halves = kind === 'grocery' ? body.split(/\s+(?:and|&)\s+/) : [body];
    if (halves.length > 1 && halves.every(h => kindOf(h, ctx) === 'grocery')) halves.forEach(h => push(h, 'grocery'));
    else push(body, kind);
  });
  return out;
}

// ---------- grocery rhythm ----------
// Every purchase date is kept on the record (hist), so the app can learn "Gu about every 2 weeks".
// A set rhythm (every, in days) wins over the learned one. Returns days, or null when there isn't enough to go on.
function cadence(r) {
  if (r.every) return r.every;
  const h = [...new Set((r.hist || []).map(t => Math.floor(t / 864e5)))].sort((a, b) => a - b);
  if (h.length < 3) return null; // two gaps at least, or one odd week looks like a rhythm
  const gaps = h.slice(1).map((d, i) => d - h[i]).filter(g => g >= 2).sort((a, b) => a - b);
  if (!gaps.length) return null;
  return gaps[Math.floor(gaps.length / 2)];
}
// Days until it's due again (negative = overdue), or null.
function dueIn(r, now = Date.now()) {
  const c = cadence(r), last = r.last || Math.max(0, ...(r.hist || []));
  if (!c || !last) return null;
  return Math.round((last + c * 864e5 - now) / 864e5);
}
function rhythmText(days) {
  if (!days) return '';
  if (days % 7 === 0 || days >= 12) { const w = Math.round(days / 7); return w <= 1 ? 'every week' : w === 4 ? 'about monthly' : `every ${w} weeks`; }
  return `every ${days} days`;
}

// Google Calendar "add event" link, prefilled (no account setup needed).
function calendarLink(t) {
  if (!t.due) return null;
  const d = t.due.replace(/-/g, '');
  let dates;
  if (t.time) {
    const s = new Date(`${t.due}T${t.time}:00`), e = new Date(s.getTime() + 30 * 6e4);
    const f = x => `${x.getFullYear()}${pad(x.getMonth() + 1)}${pad(x.getDate())}T${pad(x.getHours())}${pad(x.getMinutes())}00`;
    dates = `${f(s)}/${f(e)}`;
  } else {
    dates = `${d}/${ymd(at(new Date(t.due + 'T12:00'), 1)).replace(/-/g, '')}`;
  }
  const p = new URLSearchParams({ action: 'TEMPLATE', text: t.t, dates, details: (t.note ? t.note + '\n\n' : '') + 'From Baro\'s Biryani tasks' });
  if (t.rep) p.set('recur', `RRULE:FREQ=${{ day: 'DAILY', week: 'WEEKLY', month: 'MONTHLY', year: 'YEARLY' }[t.rep.unit]};INTERVAL=${t.rep.n}`);
  try { p.set('ctz', Intl.DateTimeFormat().resolvedOptions().timeZone); } catch {}
  return 'https://calendar.google.com/calendar/render?' + p.toString();
}

}

  for (const k of Object.keys(LIST_NAMES)) {
    if (ids[k] && have.some(l => l.id === ids[k])) continue;
    const found = have.find(l => l.title === LIST_NAMES[k]);
    ids[k] = found ? found.id : Tasks.Tasklists.insert({ title: LIST_NAMES[k] }).id;
  }
  P_.setProperty('LISTS', JSON.stringify(ids));
  return ids;
}
// Per-item link between an app record and its Google task / calendar event: property "m:<appId>" = {g, l, e}.
const getMap_ = id => JSON.parse(P_.getProperty('m:' + id) || 'null');
const setMap_ = (id, m) => (m ? P_.setProperty('m:' + id, JSON.stringify(m)) : P_.deleteProperty('m:' + id));
function allMaps_() {
  const out = {}, all = P_.getProperties();
  for (const k in all) if (k.indexOf('m:') === 0) out[k.slice(2)] = JSON.parse(all[k]);
  return out;
}
function listTasks_(listId) {
  let out = [], pageToken;
  do {
    const r = Tasks.Tasks.list(listId, { maxResults: 100, showCompleted: true, showHidden: true, pageToken });
    out = out.concat(r.items || []); pageToken = r.nextPageToken;
  } while (pageToken);
  return out;
}

// Where an app record belongs in Google: 'todo' | 'groc' | null (not shown).
function placeOf_(it) {
  if (it.del) return null;
  if (it.kind === 'grocery') return it.gone ? null : 'groc';
  return (it.list || 'todo') === 'todo' ? 'todo' : null;
}
function resourceOf_(it) {
  const aisle = it.kind === 'grocery' ? (AISLES.find(a => a.k === (it.aisle || 'other')) || {}).label : '';
  // The widget has no colors, so tasks lead with their category emoji, and pinned ones with a star.
  const title = it.kind === 'grocery' ? it.t + (it.qty ? ' ×' + it.qty : '') + (it.store ? ' · ' + storeLabel(it.store) : '')
    : (it.pri ? '⭐ ' : '') + (it.emoji && it.emoji !== '•' ? it.emoji + ' ' : '') + it.t + (it.time ? ' · ' + fmtTime_(it.time) : '');
  const r = { title, notes: it.kind === 'grocery' ? aisle : (it.note || ''), status: it.done ? 'completed' : 'needsAction' };
  r.due = it.kind === 'task' && it.due ? it.due + 'T00:00:00.000Z' : null;
  if (!it.done) r.completed = null;
  return r;
}

function sync_(items, tz) {
  const ids = listIds_();
  // 1. Push the app's changes.
  for (const it of items) {
    if (!it || !it.id) continue;
    let m = getMap_(it.id);
    const place = placeOf_(it);
    if (m && (!place || m.l !== place)) { try { Tasks.Tasks.remove(ids[m.l], m.g); } catch (e) {} m.g = null; }
    if (place) {
      const res = resourceOf_(it);
      let done = false;
      if (m && m.g) { try { Tasks.Tasks.patch(res, ids[place], m.g); done = true; } catch (e) {} }
      if (!done) { const g = Tasks.Tasks.insert(res, ids[place]); m = Object.assign(m || {}, { g: g.id }); }
      m.l = place;
      if (it.pri && !it.done) { try { Tasks.Tasks.move(ids[place], m.g); } catch (e) {} } // pinned: top of the widget
    }
    // Timed tasks also go on the calendar, so the reminder rings at the right time.
    const wantEvent = it.kind === 'task' && !it.del && !it.done && it.due && it.time && (it.list || 'todo') !== 'watch';
    m = m || {};
    m.e = calendar_(it, wantEvent ? m.e : null, wantEvent, tz);
    setMap_(it.id, m.g || m.e ? m : null);
  }
  // 2. Report back what changed on the Google side (checked off or deleted in the widget, or added there).
  const maps = allMaps_(), byG = {};
  for (const id in maps) if (maps[id].g) byG[maps[id].g] = id;
  const state = [], inbox = [];
  for (const l of ['todo', 'groc']) {
    const seen = {};
    for (const g of listTasks_(ids[l])) {
      if (g.deleted) continue;
      const id = byG[g.id];
      if (id) { seen[g.id] = 1; state.push({ id, done: g.status === 'completed' }); continue; }
      if (!g.title || g.status === 'completed') continue;
      // New from the widget or by voice: give it an app id and link it now.
      const nid = 'g' + g.id.replace(/[^A-Za-z0-9]/g, '').slice(-14);
      const time = P_.getProperty('t:' + g.id);
      setMap_(nid, { g: g.id, l }); P_.deleteProperty('s:' + g.id); P_.deleteProperty('t:' + g.id);
      inbox.push({ id: nid, list: l, title: g.title, notes: g.notes || '', due: g.due ? g.due.slice(0, 10) : null, time });
    }
    for (const id in maps) if (maps[id].l === l && maps[id].g && !seen[maps[id].g] && !inbox.some(x => x.id === id)) { state.push({ id, gone: true }); setMap_(id, maps[id].e ? { e: maps[id].e } : null); }
  }
  return { ok: true, state, inbox };
}

function calendar_(it, eventId, want, tz) {
  const cal = CalendarApp.getDefaultCalendar();
  let ev = null;
  if (eventId) { try { ev = cal.getEventById(eventId); } catch (e) {} }
  if (!want) { if (ev) { try { ev.deleteEvent(); } catch (e) {} } return null; }
  const start = Utilities.parseDate(it.due + ' ' + it.time, tz || Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm');
  const end = new Date(start.getTime() + 30 * 60e3);
  if (ev) { ev.setTitle(it.t); ev.setTime(start, end); } else ev = cal.createEvent(it.t, start, end, { description: "From Baro's Biryani tasks" });
  ev.removeAllReminders();
  if (typeof it.rm === 'number' && it.rm >= 0) ev.addPopupReminder(it.rm);
  else if (it.rm === 'morning') ev.addPopupReminder(Math.max(0, Math.round((start.getHours() * 60 + start.getMinutes()) - 9 * 60)));
  return ev.getId();
}

// Every minute: clean up things added from the widget or by voice before the app sees them.
// "milk" typed into To do moves to Groceries; "call dad tomorrow 5pm" gets a due date and a clean title;
// a grocery that's already on the list is removed instead of duplicated.
function tidy() {
  const ids = listIds_(), maps = allMaps_(), known = {};
  for (const id in maps) if (maps[id].g) known[maps[id].g] = 1;
  // "Hey Google, remind me to..." lands in Google's default "My Tasks" list: move those into Baro's To do.
  // Only undated ones or ones due by today: a repeating Google task makes its next copy when this one is removed,
  // and that next copy is dated in the future, so it waits in My Tasks until it's due (no endless copying).
  const today = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd');
  try {
    const open = listTasks_(ids.todo).filter(g => !g.deleted && g.status !== 'completed');
    for (const g of listTasks_('@default')) {
      if (g.deleted || g.status === 'completed' || !g.title) continue;
      if (g.due && g.due.slice(0, 10) > today) continue;
      if (!open.some(x => x.title === g.title && (x.due || '').slice(0, 10) === (g.due || '').slice(0, 10))) open.push(Tasks.Tasks.insert({ title: g.title, notes: g.notes, due: g.due }, ids.todo));
      Tasks.Tasks.remove('@default', g.id);
    }
  } catch (e) {}
  dedupeTodo_(ids, known);
  const groc = listTasks_(ids.groc).filter(g => !g.deleted && g.status !== 'completed');
  for (const g of listTasks_(ids.todo)) {
    if (g.deleted || g.status === 'completed' || known[g.id] || !g.title || P_.getProperty('s:' + g.id)) continue;
    const parts = splitCapture(g.title);
    let first = true;
    for (const p of parts) {
      if (p.kind === 'grocery') {
        const bits = groceryBits(p.text);
        if (!groc.some(x => sameItem(x.title.split(' ×')[0].split(' · ')[0], bits.name))) groc.push(Tasks.Tasks.insert({ title: bits.name + (bits.qty ? ' ×' + bits.qty : ''), notes: (AISLES.find(a => a.k === aisleOf(bits.name)) || {}).label }, ids.groc));
      } else {
        const w = parseWhen(p.text);
        const res = { title: w.title[0].toUpperCase() + w.title.slice(1) };
        if (w.due && !g.due) res.due = w.due + 'T00:00:00.000Z';
        // Google Tasks only stores a date, so a spoken time ("5pm") is kept here until the app picks it up.
        let gid = g.id;
        if (first) { Tasks.Tasks.patch(res, ids.todo, g.id); first = false; } else gid = Tasks.Tasks.insert(res, ids.todo).id;
        P_.setProperty('s:' + gid, '1');
        if (w.time) P_.setProperty('t:' + gid, w.time);
        continue;
      }
    }
    if (first) Tasks.Tasks.remove(ids.todo, g.id); // everything in it was groceries
  }
  // Groceries list: drop new duplicates.
  const seen = [];
  for (const g of listTasks_(ids.groc)) {
    if (g.deleted || g.status === 'completed' || !g.title) continue;
    const name = g.title.split(' ×')[0].split(' · ')[0];
    if (!known[g.id] && seen.some(n => sameItem(n, name))) { Tasks.Tasks.remove(ids.groc, g.id); continue; }
    seen.push(name);
  }
}

// Same title more than once in Baro's To do (and not linked to an app task): keep the one due first.
// This also clears the copies an older version of this script made from a repeating task in My Tasks.
function dedupeTodo_(ids, known) {
  const keep = {};
  const items = listTasks_(ids.todo).filter(g => !g.deleted && g.status !== 'completed' && g.title && !known[g.id])
    .sort((a, b) => (a.due || '9999').localeCompare(b.due || '9999'));
  for (const g of items) {
    const k = g.title.trim().toLowerCase();
    if (keep[k]) { try { Tasks.Tasks.remove(ids.todo, g.id); } catch (e) {} } else keep[k] = 1;
  }
}
